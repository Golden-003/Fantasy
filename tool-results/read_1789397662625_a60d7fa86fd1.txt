    56→    56→// Facteurs (affichés à l'utilisateur) : domicile, adversaire promu,
    57→    57→// adversaire fort, série officiellement clémente du club.
    58→    58→function fixtureDifficulty(club: string, opponent: string, isHome: boolean): { difficulty: Difficulty; note: string } {
    59→    59→  if (!TRACKED.has(club)) return { difficulty: 'INCONNU', note: 'Fixture non publiée dans l’article officiel R4-8' }
    60→    60→  let ease = 0
    61→    61→  const factors: string[] = []
    62→    62→  if (isHome) { ease += 1; factors.push('à domicile') }
    63→    63→  else { factors.push('à l’extérieur') }
    64→    64→  if (PROMOTED_OFFICIAL.has(opponent)) { ease += 1.5; factors.push(`adversaire promu (${opponent})`) }
    65→    65→  else if (PROMOTED_DEDUCE.has(opponent)) { ease += 1.25; factors.push(`adversaire promu (${opponent}, déduit des fixtures)`) }
    66→    66→  if (STRONG.has(opponent)) { ease -= 1; factors.push(`adversaire fort (${opponent})`) }
    67→    67→  if (TRACKED.has(club)) { ease += 0.5; factors.push('série R4-8 officiellement parmi les 5 plus clémentes (article)') }
    68→    68→  const difficulty: Difficulty = ease >= 1.5 ? 'FACILE' : ease >= 0.5 ? 'MOYEN' : 'DIFFICILE'
    69→    69→  return { difficulty, note: factors.join(' · ') }
    70→    70→}

   404→   404→export async function saveLive(round: number, input: LiveSaveInput): Promise<LiveView> {
   405→   405→  if (!Number.isInteger(round) || round < 1 || round > 38) throw new Error('Journée invalide')
   406→   406→  const now = new Date().toISOString()
   407→   407→
   408→   408→  // État de la journée (capitaine / token)
   409→   409→  await db.liveRound.upsert({
   410→   410→    where: { round },
   411→   411→    update: {
   412→   412→      captainPlayerId: input.captainPlayerId ?? undefined,
   413→   413→      tripleCaptain: input.tripleCaptain ?? undefined,
   414→   414→      updatedAt: now,
   415→   415→    },
   416→   416→    create: {
   417→   417→      round,
   418→   418→      captainPlayerId: input.captainPlayerId ?? null,
   419→   419→      tripleCaptain: input.tripleCaptain ?? false,
   420→   420→      updatedAt: now,
   421→   421→    },
   422→   422→  })
   423→   423→
   424→   424→  // Saisies de points
   425→   425→  for (const e of input.entries ?? []) {
   426→   426→    const data = { round, playerId: e.playerId, points: e.points, note: e.note ?? null, updatedAt: now }
   427→   427→    await db.liveEntry.upsert({
   428→   428→      where: { round_playerId: { round, playerId: e.playerId } },
   429→   429→      update: { points: e.points, note: e.note ?? null, updatedAt: now },
   430→   430→      create: data,
   431→   431→    })
   432→   432→  }
   433→   433→
   434→   434→  // Journal : 1 événement max / 10 min par journée (évite le spam d'autosave)
   435→   435→  const tenMinAgo = new Date(Date.now() - 10 * 60 * 1000).toISOString()
   436→   436→  const recent = await db.dataEvent.findFirst({
   437→   437→    where: { kind: 'SAISIE', detail: { contains: `R${round} ·` }, at: { gte: tenMinAgo } },
   438→   438→  })
   439→   439→  if (!recent) {
   440→   440→    await db.dataEvent.create({
   441→   441→      data: {
   442→   442→        at: now,
   443→   443→        kind: 'SAISIE',
   444→   444→        title: 'Saisie live — Match Center',
   445→   445→        detail: `R${round} · points recopiés depuis l'app Sofascore par l'utilisateur pendant ou après les matchs (canal temps réel V1).`,
   446→   446→      },
   447→   447→    })
   448→   448→  }
   449→   449→
   450→   450→  return getLive(round)
   451→   451→}
