    55→// ── Difficulté de fixture — heuristique TRANSPARENTE ───────────
    56→// Facteurs (affichés à l'utilisateur) : domicile, adversaire promu,
    57→// adversaire fort, série officiellement clémente du club.
    58→function fixtureDifficulty(club: string, opponent: string, isHome: boolean): { difficulty: Difficulty; note: string } {
    59→  if (!TRACKED.has(club)) return { difficulty: 'INCONNU', note: 'Fixture non publiée dans l’article officiel R4-8' }
    60→  let ease = 0
    61→  const factors: string[] = []
    62→  if (isHome) { ease += 1; factors.push('à domicile') }
    63→  else { factors.push('à l’extérieur') }
    64→  if (PROMOTED_OFFICIAL.has(opponent)) { ease += 1.5; factors.push(`adversaire promu (${opponent})`) }
    65→  else if (PROMOTED_DEDUCE.has(opponent)) { ease += 1.25; factors.push(`adversaire promu (${opponent}, déduit des fixtures)`) }
    66→  if (STRONG.has(opponent)) { ease -= 1; factors.push(`adversaire fort (${opponent})`) }
    67→  if (TRACKED.has(club)) { ease += 0.5; factors.push('série R4-8 officiellement parmi les 5 plus clémentes (article)') }
    68→  const difficulty: Difficulty = ease >= 1.5 ? 'FACILE' : ease >= 0.5 ? 'MOYEN' : 'DIFFICILE'
    69→  return { difficulty, note: factors.join(' · ') }
    70→}
    71→
    72→const easeBonus = (d: Difficulty) => (d === 'FACILE' ? 2 : d === 'MOYEN' ? 1 : 0)

    77→export async function getTeam(): Promise<TeamView> {
    78→  const slots = await db.squadSlot.findMany({
    79→    include: { player: true },
    80→    orderBy: [{ role: 'asc' }, { slotPosition: 'asc' }],
    81→  })
    82→  const fixturesR5 = await db.fixture.findMany({ where: { round: 5 } })
    83→  const fixMap = new Map(fixturesR5.map((f) => [f.club, f]))
    84→
    85→  const view = (s: (typeof slots)[number]): SquadPlayerView => {
    86→    const fx = s.player.clubConfirmed ? fixMap.get(s.player.club) : undefined
    87→    let fixtureR5: FixtureLite | null = null
    88→    if (fx) {
    89→      const { difficulty, note } = fixtureDifficulty(fx.club, fx.opponent, fx.isHome)
    90→      fixtureR5 = { opponent: fx.opponent, isHome: fx.isHome, difficulty, note }
    91→    }
    92→    return {
    93→      id: s.player.id,
    94→      name: s.player.name,
    95→      club: s.player.club,
    96→      clubConfirmed: s.player.clubConfirmed,
    97→      position: s.slotPosition as Pos,
    98→      role: s.role as 'TITULAIRE' | 'BANC',
    99→      captain: s.captain,
   100→      pointsR4: s.pointsR4,
   101→      pointsNote: s.pointsNote,
   102→      price: s.player.price,
   103→      ownership: s.player.ownership,
   104→      priceSource: s.player.priceSource,
   105→      formNote: s.player.formNote,
   106→      fixtureR5,
   107→    }
   108→  }
   109→
   110→  const starters = slots.filter((s) => s.role === 'TITULAIRE').map(view)
   111→  const bench = slots.filter((s) => s.role === 'BANC').map(view)
   112→
   113→  const byPos = (arr: SquadPlayerView[]) => {
   114→    const g = arr.filter((p) => p.position === 'G').length
   115→    const d = arr.filter((p) => p.position === 'D').length
   116→    const m = arr.filter((p) => p.position === 'M').length
   117→    const a = arr.filter((p) => p.position === 'A').length
   118→    return [g, d, m, a]
   119→  }
   120→  const [, d, m, a] = byPos(starters)
   121→  const formation = `${d}-${m}-${a}`
   122→
   123→  const priced = [...starters, ...bench].filter((p) => p.price != null)
   124→  const knownSpend = priced.reduce((acc, p) => acc + (p.price ?? 0), 0)
   125→  const unknown = [...starters, ...bench].filter((p) => p.price == null).map((p) => p.name)
   126→
   127→  const scores = await db.roundScore.findMany({ include: { manager: true }, orderBy: { round: 'asc' } })
   128→  const mine = scores.filter((s) => s.manager.isUser && s.points != null)
   129→  const at = (r: number) => mine.find((s) => s.round === r)?.points ?? 0
   130→  const totals = {
   131→    r1: at(1), r2: at(2), r3: at(3), r4: at(4),
   132→    total: mine.find((s) => s.round === 4)?.totalAfter ?? mine.reduce((acc, s) => acc + (s.points ?? 0), 0),
   133→  }
   134→
   135→  return {
   136→    formation,
   137→    starters,
   138→    bench,
   139→    knownSpend: Math.round(knownSpend * 10) / 10,
   140→    knownPriceCount: priced.length,
   141→    unknownPriceCount: unknown,
   142→    totals,
   143→  }
   144→}

   147→export async function getCaptainPicks(team?: TeamView): Promise<CaptainPick[]> {
   148→  const t = team ?? (await getTeam())
   149→  const picks: Omit<CaptainPick, 'rank'>[] = []
   150→  for (const p of t.starters) {
   151→    if (!p.fixtureR5 || p.fixtureR5.difficulty === 'INCONNU') continue
   152→    const reasons: string[] = []
   153→    let score = 0
   154→    if (p.pointsR4 != null) {
   155→      score += p.pointsR4
   156→      reasons.push(`${p.pointsR4} pts réels en R4 (capture)`)
   157→    } else if (p.pointsNote) {
   158→      reasons.push(`points R4 incomplets : ${p.pointsNote}`)
   159→    }
   160→    if (p.formNote) {
   161→      score += 1
   162→      reasons.push(`forme officielle : ${p.formNote}`)
   163→    }
   164→    if (p.captain) {
   165→      score += 0.5
   166→      reasons.push('capitaine sortant (R4)')
   167→    }
   168→    score += easeBonus(p.fixtureR5.difficulty)
   169→    reasons.push(`R5 ${fixtureLabel(p.fixtureR5)} — ${p.fixtureR5.difficulty} (${p.fixtureR5.note})`)
   170→    if (p.ownership != null) reasons.push(`détenu par ${p.ownership}% du jeu (article officiel)`)
   171→    picks.push({
   172→      playerId: p.id, name: p.name, club: p.club, position: p.position,
   173→      fixture: fixtureLabel(p.fixtureR5), difficulty: p.fixtureR5.difficulty,
   174→      score: Math.round(score * 10) / 10, reasons,
   175→    })
   176→  }
   177→  picks.sort((x, y) => y.score - x.score)
   178→  return picks.map((p, i) => ({ ...p, rank: i + 1 }))
   179→}

   200→export async function getMarketTargets(): Promise<MarketTarget[]> {
   201→  const t = await getTeam()
   202→  const squadIds = new Set([...t.starters, ...t.bench].map((p) => p.id))
   203→  const players = await db.player.findMany({ where: { price: { not: null } } })
   204→  const fixturesR5 = new Map((await db.fixture.findMany({ where: { round: 5 } })).map((f) => [f.club, f]))
   205→  const targets = players
   206→    .filter((p) => !squadIds.has(p.id))
   207→    .map((p) => {
   208→      const fx = p.clubConfirmed ? fixturesR5.get(p.club) : undefined
   209→      const dif = fx ? fixtureDifficulty(fx.club, fx.opponent, fx.isHome) : null
   210→      let rationale = ''
   211→      if (fx && dif) rationale = `R5 ${fixtureLabel({ opponent: fx.opponent, isHome: fx.isHome })} — ${dif.difficulty}`
   212→      else rationale = 'Fixture R5 non publiée'
   213→      return {
   214→        playerId: p.id, name: p.name, club: p.club, position: p.position as Pos,
   215→        price: p.price as number, ownership: p.ownership ?? 0, formNote: p.formNote, rationale,
   216→      }
   217→    })
   218→  const rankOrder = ['FACILE', 'MOYEN', 'DIFFICILE', 'INCONNU']
   219→  targets.sort((a, b) => {
   220→    const da = rankOrder.indexOf(a.rationale.split(' — ')[1]?.split(' ')[0] ?? 'INCONNU')
   221→    const dbv = rankOrder.indexOf(b.rationale.split(' — ')[1]?.split(' ')[0] ?? 'INCONNU')
   222→    return da - dbv || b.ownership - a.ownership
   223→  })
   224→  return targets
   225→}

   328→export async function getLive(round: number): Promise<LiveView> {
   329→  const [team, entries, liveRound, league] = await Promise.all([
   330→    getTeam(),
   331→    db.liveEntry.findMany({ where: { round }, include: { player: true } }),
   332→    db.liveRound.findUnique({ where: { round } }),
   333→    getLeague(),
   334→  ])
   335→
   336→  const entryByPlayer = new Map(entries.map((e) => [e.playerId, e]))
   337→  const rowOf = (p: SquadPlayerView): LivePlayerRow => {
   338→    const e = entryByPlayer.get(p.id)
   339→    const fx = p.fixtureR5 ? `${p.fixtureR5.isHome ? 'vs' : '@'} ${p.fixtureR5.opponent}` : null
   340→    return {
   341→      playerId: p.id, name: p.name, club: p.club, clubConfirmed: p.clubConfirmed,
   342→      position: p.position, role: p.role, fixture: fx, pointsR4: p.pointsR4,
   343→      points: e?.points ?? null, updatedAt: e?.updatedAt ?? null,
   344→    }
   345→  }
   346→  const rows: LivePlayerRow[] = [...team.starters, ...team.bench].map(rowOf)
   347→
   348→  const multiplier: 2 | 3 = liveRound?.tripleCaptain ? 3 : 2
   349→  const captainPlayerId = liveRound?.captainPlayerId ?? null
   350→  const captainEntry = captainPlayerId ? entryByPlayer.get(captainPlayerId) : undefined
   351→  const captainIsStarter = rows.some((r) => r.playerId === captainPlayerId && r.role === 'TITULAIRE')
   352→  const captainPoints = captainIsStarter && captainEntry?.points != null ? captainEntry.points : 0
   353→  const captainName = rows.find((r) => r.playerId === captainPlayerId)?.name ?? null
   354→
   355→  const starterRows = rows.filter((r) => r.role === 'TITULAIRE')
   356→  const benchRows = rows.filter((r) => r.role === 'BANC')
   357→  const startersEntered = starterRows.filter((r) => r.points != null).length
   358→  const startersPoints = starterRows.reduce((acc, r) => acc + (r.points ?? 0), 0)
   359→  const captainBonus = (multiplier - 1) * captainPoints
   360→  const benchPoints = benchRows.reduce((acc, r) => acc + (r.points ?? 0), 0)
   361→  const liveRoundPoints = startersPoints + captainBonus
   362→  const baseTotal = team.totals.total
   363→  const updates = entries.map((e) => e.updatedAt).filter(Boolean).sort()
   364→  const lastUpdate = updates.length ? updates[updates.length - 1] : null
   365→
   366→  // Classement simulé : mes points live s'ajoutent à mon total archivé,
   367→  // rivaux figés à leur dernier total connu (captures).
   368→  const officialRank = new Map(league.standings.map((s) => [s.name, s.rank]))
   369→  const sim: LiveStandingRow[] = league.standings.map((s) => ({
   370→    name: s.name,
   371→    isUser: s.isUser,
   372→    baseTotal: s.total,
   373→    liveRoundPoints: s.isUser ? liveRoundPoints : null,
   374→    projectedTotal: s.isUser ? baseTotal + liveRoundPoints : s.total,
   375→    rank: 0,
   376→    moved: false,
   377→  }))
   378→  sim.sort((a, b) => (b.projectedTotal ?? b.baseTotal ?? 0) - (a.projectedTotal ?? a.baseTotal ?? 0))
   379→  sim.forEach((r, i) => { r.rank = i + 1; r.moved = officialRank.get(r.name) !== r.rank })
   380→
   381→  return {
   382→    round,
   383→    roundDate: ROUND_DATES[round] ?? null,
   384→    captainPlayerId,
   385→    captainName,
   386→    tripleCaptain: liveRound?.tripleCaptain ?? false,
   387→    multiplier,
   388→    rows,
   389→    live: {
   390→      startersEntered, startersTotal: starterRows.length,
   391→      startersPoints, captainBonus, benchPoints, liveRoundPoints,
   392→      baseTotal, projectedTotal: baseTotal + liveRoundPoints, lastUpdate,
   393→    },
   394→    standings: sim,
   395→  }
   396→}

   406→  const now = new Date().toISOString()
   407→
   408→  // État de la journée (capitaine / token)
   409→  await db.liveRound.upsert({
   410→    where: { round },
   411→    update: {
   412→      captainPlayerId: input.captainPlayerId ?? undefined,
   413→      tripleCaptain: input.tripleCaptain ?? undefined,
   414→      updatedAt: now,
   415→    },
   416→    create: {
   417→      round,
   418→      captainPlayerId: input.captainPlayerId ?? null,
   419→      tripleCaptain: input.tripleCaptain ?? false,
   420→      updatedAt: now,
   421→    },
   422→  })
   423→
   424→  // Saisies de points
   425→  for (const e of input.entries ?? []) {
   426→    const data = { round, playerId: e.playerId, points: e.points, note: e.note ?? null, updatedAt: now }
   427→    await db.liveEntry.upsert({
   428→      where: { round_playerId: { round, playerId: e.playerId } },
   429→      update: { points: e.points, note: e.note ?? null, updatedAt: now },
   430→      create: data,
   431→    })
   432→  }
   433→
   434→  // Journal : 1 événement max / 10 min par journée (évite le spam d'autosave)
   435→  const tenMinAgo = new Date(Date.now() - 10 * 60 * 1000).toISOString()
   436→  const recent = await db.dataEvent.findFirst({
   437→    where: { kind: 'SAISIE', detail: { contains: `R${round} ·` }, at: { gte: tenMinAgo } },
   438→  })
   439→  if (!recent) {
   440→    await db.dataEvent.create({
   441→      data: {
   442→        at: now,
   443→        kind: 'SAISIE',
   444→        title: 'Saisie live — Match Center',
   445→        detail: `R${round} · points recopiés depuis l'app Sofascore par l'utilisateur pendant ou après les matchs (canal temps réel V1).`,
   446→      },
   447→    })
   448→  }
   449→
   450→  return getLive(round)
   451→}
